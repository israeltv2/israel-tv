// landing-en
<script>try {
const theme = localStorage.getItem('iltv-theme');
if (theme) document.documentElement.dataset.theme = theme;
} catch {}</script>
<script>(() => {
const d = document;
const root = d.documentElement;
const lang = root.lang || 'en';
d.getElementById('theme')?.addEventListener('click', () => {
const next = root.dataset.theme === 'light' ? 'dark' : 'light';
const apply = () => {
root.dataset.theme = next;
};
if (d.startViewTransition && !matchMedia('(prefers-reduced-motion: reduce)').matches)
d.startViewTransition(apply);
else apply();
try {
localStorage.setItem('iltv-theme', next);
} catch {}
});
const relative = window.Intl?.RelativeTimeFormat
? new Intl.RelativeTimeFormat(lang, { numeric: 'auto' })
: null;
const full = new Intl.DateTimeFormat(lang, { dateStyle: 'medium', timeStyle: 'short' });
const UNITS = [
[60, 'second', 1],
[3600, 'minute', 60],
[86400, 'hour', 3600],
[2592000, 'day', 86400],
[Number.POSITIVE_INFINITY, 'month', 2592000],
];
const ago = (t) => {
const s = (t - Date.now()) / 1000;
const unit = UNITS.find(([limit]) => Math.abs(s) < limit);
return relative.format(Math.round(s / unit[2]), unit[1]).replace(/\s*\(\d+\)$/, '');
};
for (const el of d.querySelectorAll('time[datetime]')) {
if (el.hasAttribute('data-abs')) continue;
let t = Date.parse(el.getAttribute('datetime'));
if (Number.isNaN(t)) continue;
if (el.hasAttribute('data-daily')) while (t < Date.now()) t += 864e5;
if (el.hasAttribute('data-future') && t < Date.now()) {
el.textContent = lang === 'he' ? 'בקרוב' : 'soon';
continue;
}
el.title = full.format(t);
if (relative) el.textContent = ago(t);
}
for (const btn of d.querySelectorAll('[data-copy]')) {
btn.addEventListener('click', () => {
const source = d.getElementById(btn.dataset.copy);
const label = btn.querySelector('span');
const done = () => {
const before = btn.dataset.label || label.textContent;
btn.dataset.label = before;
label.textContent = btn.dataset.done;
setTimeout(() => {
label.textContent = before;
}, 1800);
};
if (navigator.clipboard) navigator.clipboard.writeText(source.textContent.trim()).then(done, () => {});
else {
const range = d.createRange();
range.selectNodeContents(source);
getSelection().removeAllRanges();
getSelection().addRange(range);
}
});
}
const manifest = `${location.origin}/manifest.json`;
for (const a of d.querySelectorAll('[data-stremio]'))
if (!a.getAttribute('href').startsWith('stremio://')) a.href = `stremio://${location.host}/manifest.json`;
for (const a of d.querySelectorAll('[data-stremio-web]'))
if (!a.href.includes('?addon='))
a.href = `https://web.stremio.com/#/addons?addon=${encodeURIComponent(manifest)}`;
for (const el of d.querySelectorAll('[data-origin]'))
if (el.textContent.startsWith('{{') || !/^https?:/.test(el.textContent)) el.textContent = manifest;
for (const tick of d.querySelectorAll('.tick')) {
const copy = tick.firstElementChild.cloneNode(true);
copy.setAttribute('aria-hidden', 'true');
for (const a of copy.querySelectorAll('a')) a.tabIndex = -1;
tick.appendChild(copy);
tick.classList.add('on');
}
const air = () => {
const now = Date.now();
for (const guide of d.querySelectorAll('[data-guide]')) {
let shown = 0;
for (const item of guide.querySelectorAll('[data-s]')) {
const start = +item.dataset.s;
const end = +item.dataset.e;
const ended = end <= now;
const on = !ended && start <= now;
item.hidden = ended || shown >= (+guide.dataset.guide || 2);
item.classList.toggle('now', on);
if (!item.hidden) shown++;
if (on) item.style.setProperty('--p', Math.min(1, (now - start) / (end - start)).toFixed(3));
}
guide.classList.toggle('empty', !shown);
}
};
air();
setInterval(air, 30000);
})();</script>

}

<script>(() => {
const FINALS = { ך: 'כ', ם: 'מ', ן: 'נ', ף: 'פ', ץ: 'צ' };
const PREFIXES = 'הובלמשכ';
const normalize = (s) =>
s
.normalize('NFKD')
.replace(/[֑-ׇ]/g, '')
.replace(/[ךםןףץ]/g, (c) => FINALS[c])
.toLowerCase()
.replace(/[׳״'"`]/g, '')
.replace(/[^\p{L}\p{N}]+/gu, ' ')
.trim();
/** A word without its one-letter prefix ("השף" → "שף"), or '' when it has none or too little would be left. */
const unprefixed = (w) => {
const n = w[0] === 'ו' && w.length > 3 && PREFIXES.includes(w[1]) ? 2 : 1;
return w.length - n >= 2 && PREFIXES.includes(w[0]) ? w.slice(n) : '';
};
/** 1: every query word starts a title word; 2: only through a prefix letter; 0: no match. */
const score = (key, words) => {
const titleWords = key.split(' ');
const bare = titleWords.map(unprefixed).filter(Boolean);
let best = 1;
for (const w of words) {
if (titleWords.some((k) => k.startsWith(w))) continue;
const w2 = unprefixed(w);
if (bare.some((k) => k.startsWith(w)) || (w2 && titleWords.some((k) => k.startsWith(w2)))) {
best = 2;
continue;
}
return 0;
}
return best;
};
const plain = (s) => s.replace(/[‎‏‪-‮⁦-⁩]/g, '');
const sized = (u) =>
/^https:\/\/media\d*\.reshet\.tv\/image\/upload\//.test(u) && !/\/upload\/[a-z]_/.test(u)
? u.replace('/image/upload/', '/image/upload/w_220,c_limit,f_auto,q_auto/')
: u;
const d = document;
const input = d.getElementById('q');
const results = d.getElementById('results');
const info = d.getElementById('info');
const empty = d.getElementById('empty');
let rows = null;
const tile = ([id, type, , name, , poster]) => {
const live = type === 'tv';
const a = d.createElement('a');
a.className = 'tile';
a.href = live ? `/live/#${id}` : `/show/${id}/`;
const image = poster ? sized(poster.replace('{{israel-tv-origin}}', '')) : '';
if (image) {
const img = d.createElement('img');
Object.assign(img, {
className: 'poster',
src: image,
alt: '',
loading: 'lazy',
width: 220,
height: 330,
});
if (live) img.style.cssText = 'object-fit:contain;padding:18%';
a.append(img);
} else {
const blank = d.createElement('span');
blank.className = 'poster blank';
blank.setAttribute('aria-hidden', 'true');
a.append(blank);
}
const title = d.createElement('span');
title.className = 't';
title.dir = 'auto';
title.textContent = plain(name);
a.append(title);
return a;
};
const run = () => {
const q = normalize(input.value);
history.replaceState(null, '', q ? `?q=${encodeURIComponent(input.value)}` : location.pathname);
empty.hidden = true;
if (!q) {
results.replaceChildren();
info.textContent = '';
return;
}
if (!rows) {
info.textContent = 'טוען…';
return;
}
const words = q.split(' ');
const hits = rows
.map((r) => [score(r[4], words), r])
.filter(([s]) => s)
.sort((a, b) => a[0] - b[0]);
info.textContent = hits.length
? hits.length === 1
? 'נמצאה תוכנית אחת'
: `נמצאו ${hits.length} תוצאות`
: 'לא נמצא כלום. נסו לחפש בעברית, כמו שהשם כתוב באתר הערוץ.';
results.replaceChildren(...hits.slice(0, 120).map(([, r]) => tile(r)));
empty.hidden = hits.length > 0;
};
input.addEventListener('input', run);
input.value = new URLSearchParams(location.search).get('q') || '';
fetch('/search.json')
.then((r) => r.json())
.then(
(json) => {
rows = json;
run();
},
() => {
info.textContent = 'החיפוש לא נטען. נסו לרענן את הדף.';
},
);
input.focus();
run();
})();</script>
